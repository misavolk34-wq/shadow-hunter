require("dotenv").config(); const express = require("express"); const crypto = require("crypto"); const path = require("path"); const { createClient } = require("@supabase/supabase-js"); const app = express(); const PORT = process.env.PORT || 3000; const BOT_TOKEN = process.env.BOT_TOKEN; const SUPABASE_URL = process.env.SUPABASE_URL; const SUPABASE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY; if (!BOT_TOKEN) throw new Error("В .env не задан BOT_TOKEN"); if (!SUPABASE_URL) throw new Error("В .env не задан SUPABASE_URL"); if (!SUPABASE_KEY) throw new Error("В .env не задан SUPABASE_SERVICE_ROLE_KEY"); const supabase = createClient(SUPABASE_URL, SUPABASE_KEY, { auth: { persistSession: false, autoRefreshToken: false } }); app.use(express.json({ limit: "30kb" })); app.use(express.static(path.join(__dirname, "public"))); const classes = { mage: { name: "Маг", icon: "🔮", hp: 85, atk: 17, def: 3 }, fighter: { name: "Боец", icon: "⚔️", hp: 115, atk: 13, def: 7 }, archer: { name: "Стрелок", icon: "🏹", hp: 95, atk: 15, def: 5 }, tank: { name: "Танк", icon: "🛡️", hp: 145, atk: 10, def: 11 } }; const enemyNames = [ ["Теневой волк", "🐺"], ["Падший рыцарь", "💀"], ["Пожиратель маны", "👹"], ["Костяной страж", "🦇"], ["Демон врат", "👾"] ]; function emptyGame() { return { classId: null, level: 1, xp: 0, gold: 100, maxFloor: 1, floor: 1, hp: 100, inventory: [], equipped: { weapon: null, armor: null, relic: null }, enemy: null, lastAction: 0 }; } function verifyTelegramData(initData) { if (!initData) return null; const params = new URLSearchParams(initData); const receivedHash = params.get("hash"); const authDate = Number(params.get("auth_date")); if (!receivedHash || !authDate) return null; // Не принимаем данные Telegram старше суток. if (Date.now() / 1000 - authDate > 86400) return null; const entries = [...params.entries()] .filter(([key]) => key !== "hash") .sort(([a], [b]) => a.localeCompare(b)); const dataCheckString = entries .map(([key, value]) => `${key}=${value}`) .join("\n"); const secretKey = crypto .createHmac("sha256", "WebAppData") .update(BOT_TOKEN) .digest(); const calculatedHash = crypto .createHmac("sha256", secretKey) .update(dataCheckString) .digest("hex"); const expected = Buffer.from(calculatedHash, "hex"); const received = Buffer.from(receivedHash, "hex"); if (expected.length !== received.length) return null; if (!crypto.timingSafeEqual(expected, received)) return null; try { return JSON.parse(params.get("user") || "null"); } catch { return null; } } function getStats(game) { const c = classes[game.classId] || classes.fighter; const result = { maxHp: c.hp + (game.level - 1) * 14, atk: c.atk + (game.level - 1) * 3, def: c.def + Math.floor((game.level - 1) * 1.2) }; for (const slot of ["weapon", "armor", "relic"]) { const item = game.inventory.find( entry => entry.id === game.equipped[slot] ); if (!item) continue; result.maxHp += item.hp || 0; result.atk += item.atk || 0; result.def += item.def || 0; } return result; } function xpNeeded(level) { return 80 + level * 42; } function rollItem(floor) { const roll = Math.random() * 100; let rarity; if (roll < 4) { rarity = { key: "mythic", name: "Мифический", mult: 3.7 }; } else if (roll < 17) { rarity = { key: "gold", name: "Золотой", mult: 2.5 }; } else if (roll < 42) { rarity = { key: "legendary", name: "Легендарный", mult: 1.7 }; } else { rarity = { key: "common", name: "Обычный", mult: 1 }; } const slot = ["weapon", "armor", "relic"][ Math.floor(Math.random() * 3) ]; const power = Math.max( 1, Math.floor((2 + floor * 0.8) * rarity.mult) ); const slotNames = { weapon: "клинок", armor: "доспех", relic: "артефакт" }; const prefixes = ["Забытый", "Теневой", "Портальный", "Древний"]; return { id: crypto.randomUUID(), name: `${prefixes[Math.floor(Math.random() * prefixes.length)]} ${slotNames[slot]}`, slot, rarity: rarity.key, rarityName: rarity.name, atk: slot === "weapon" ? power : 0, def: slot === "armor" ? Math.max(1, Math.floor(power * 0.65)) : 0, hp: slot === "relic" ? power * 5 : 0 }; } async function fetchPlayer(telegramId) { const { data, error } = await supabase .from("players") .select("*") .eq("telegram_id", telegramId) .maybeSingle(); if (error) throw error; return data; } async function savePlayer(telegramId, name, game) { const { error } = await supabase .from("players") .upsert({ telegram_id: telegramId, name, level: game.level, xp: game.xp, game, updated_at: Date.now() }, { onConflict: "telegram_id" }); if (error) throw error; } async function auth(req, res, next) { try { const telegramUser = verifyTelegramData( req.get("x-telegram-init-data") ); if (!telegramUser?.id) { return res.status(401).json({ error: "Открой игру через Telegram." }); } const telegramId = String(telegramUser.id); const name = String( telegramUser.first_name || telegramUser.username || "Охотник" ).slice(0, 40); let player = await fetchPlayer(telegramId); if (!player) { const game = emptyGame(); await savePlayer(telegramId, name, game); player = await fetchPlayer(telegramId); } else if (player.name !== name) { await supabase .from("players") .update({ name }) .eq("telegram_id", telegramId); player.name = name; } req.player = player; req.telegramId = telegramId; req.game = player.game || emptyGame(); next(); } catch (error) { console.error("Ошибка авторизации/Supabase:", error.message); res.status(500).json({ error: "Не удалось загрузить сохранение." }); } } app.use("/api", auth); app.get("/api/state", (req, res) => { res.json({ name: req.player.name, game: req.game, stats: req.game.classId ? getStats(req.game) : null }); }); app.post("/api/class", async (req, res) => { try { const game = req.game; const classId = req.body?.classId; if (!classes[classId]) { return res.status(400).json({ error: "Неизвестный класс." }); } if (game.classId) { return res.status(400).json({ error: "Класс уже выбран." }); } game.classId = classId; game.hp = getStats(game).maxHp; await savePlayer(req.telegramId, req.player.name, game); res.json({ ok: true }); } catch (error) { console.error(error); res.status(500).json({ error: "Не удалось сохранить класс." }); } }); app.post("/api/start", async (req, res) => { try { const game = req.game; if (!game.classId) { return res.status(400).json({ error: "Сначала выбери класс." }); } if (game.enemy && game.enemy.hp > 0) { return res.json({ ok: true, enemy: game.enemy }); } const floor = Number(req.body?.floor || game.maxFloor); if (!Number.isInteger(floor) || floor < 1 || floor > game.maxFloor) { return res.status(400).json({ error: "Этот этаж ещё закрыт." }); } game.floor = floor; const [name, icon] = enemyNames[Math.floor(Math.random() * enemyNames.length)]; const boss = floor % 5 === 0; const maxHp = Math.floor((45 + floor * 22) * (boss ? 2.4 : 1)); game.enemy = { floor, name: boss ? `Босс: ${name}` : name, icon: boss ? "🐉" : icon, maxHp, hp: maxHp, atk: Math.floor(7 + floor * 2.3) }; await savePlayer(req.telegramId, req.player.name, game); res.json({ ok: true, enemy: game.enemy }); } catch (error) { console.error(error); res.status(500).json({ error: "Не удалось открыть врата." }); } }); app.post("/api/attack", async (req, res) => { try { const game = req.game; const enemy = game.enemy; if (!game.classId || !enemy || enemy.hp <= 0) { return res.status(400).json({ error: "Сначала войди во врата." }); } if (Date.now() - (game.lastAction || 0) < 450) { return res.status(429).json({ error: "Подожди мгновение." }); } game.lastAction = Date.now(); const stats = getStats(game); const special = Boolean(req.body?.special); const damage = Math.max( 1, Math.floor( stats.atk * (special ?
